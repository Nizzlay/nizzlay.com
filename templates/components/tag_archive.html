{#- A tag's posts as an archive, grouped by year, newest first, with the markup of tabi's archive page (archive.html), so it's styled the same. Used in templates (tags/single.html, tag-archive.html), not in content: taxonomies aren't available while Markdown renders. -#} {% component tag_archive(term: map, @config: map, @language_strings = {}) %} {%- set posts = term.pages | sort(attribute="date") | reverse -%} {%- set date_locale = language_strings["date_locale"] | default(value="en_GB") -%} {%- set date_format = config.extra.archive_date_format | default(value="dd MMM") -%} {#- group_by doesn't keep the posts' order, so sort the years separately. -#} {%- set posts_by_year = posts | group_by(attribute="year") -%} {%- set years = [] -%} {%- for year, ignored in posts_by_year -%} {%- set_global years = [...years, year] -%} {%- endfor -%} {%- set years = years | sort | reverse %}
    {%- for year in years %}

  • {{ year }}

      {%- for post in posts_by_year[year] %}
    • {{ post.date | date(format=date_format, locale=date_locale) }}
      {{ post.title | markdown(inline=true) | safe }}
      • {%- endfor %}
    • {%- endfor %}
{% endcomponent tag_archive %}
